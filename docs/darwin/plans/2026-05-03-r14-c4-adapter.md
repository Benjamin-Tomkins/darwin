**Goal:** Build the C4 plan format adapter (R14.1–R14.15) — a pure-function layer that lets the sparse-workflow controller ingest `index.adoc` plans with embedded Structurizr DSL, validate slugs, derive branch names, emit trailers, enforce property lifecycle, and inject sandbox constraints from `[task]` blocks.

**Architecture:** All adapter logic lives in `src/c4/`. Every exported function is pure (no filesystem or Git side-effects); the controller's existing scaffold machinery calls these functions and acts on the results. Types are shared through `src/c4/types.ts`. The controller entry point (R14.1) lives in `src/scaffold/worktree.ts`, which gains a `--c4 <plan-dir>` flag.

**Tech Stack:** TypeScript (targets plain ESM; executed at runtime by Bun, Node.js 20+, or Deno — detected by `/darwin-init` and stored in `~/.claude/darwin-state/runtime.json`); vitest 2.x for testing (requires Bun or Node; Deno is an execution-only runtime for the compiled helpers). No external AsciiDoc or Structurizr DSL libraries — targeted line-by-line parsers cover the required subset.

- [ ] **Step 1: Create `scripts/detect-runtime.sh`**

```bash
#!/usr/bin/env bash
set -euo pipefail

RUNTIME_DIR="${HOME}/.claude/darwin-state"
mkdir -p "$RUNTIME_DIR"

if command -v bun &>/dev/null; then
  printf '{"runtime":"bun","exec":"bun","pm":"bun","run_flags":[]}\n' \
    | tee "${RUNTIME_DIR}/runtime.json"
elif command -v node &>/dev/null \
  && node --version 2>/dev/null | grep -qE '^v(2[0-9]|[3-9][0-9])'; then
  printf '{"runtime":"node","exec":"node","pm":"npm","run_flags":[]}\n' \
    | tee "${RUNTIME_DIR}/runtime.json"
elif command -v deno &>/dev/null; then
  # Deno: execution runtime only. Use npm/node for build/test.
  printf '{"runtime":"deno","exec":"deno","pm":"npm","run_flags":["run","--allow-read","--allow-write","--allow-env"]}\n' \
    | tee "${RUNTIME_DIR}/runtime.json"
  echo "WARNING: Deno detected as execution runtime. npm/node still required for build and tests." >&2
else
  echo "ERROR: No supported JS runtime found. Install Bun, Node.js 20+, or Deno." >&2
  exit 1
fi
```

```typescript
// src/c4/parser.ts

export function extractDslBlock(adocContent: string): string {
  const m = adocContent.match(/\[source,structurizr\]\n-{4,}\n([\s\S]*?)\n-{4,}/);
  if (!m?.[1]) {
    throw new Error('No [source,structurizr] block found in index.adoc');
  }
  return m[1];
}

export function extractDocAttribute(adocContent: string, name: string): string | undefined {
  const m = adocContent.match(new RegExp(`^:${name}:\\s*(.*)$`, 'm'));
  const val = m?.[1]?.trim();
  return val || undefined;
}
```

```typescript
// TOP OF FILE — add this import line before the existing export functions:
import type { C4ElementType, DslElement, ParsedIndex, Phase } from './types.js';

// BOTTOM OF FILE — append these constants and functions:

const ELEMENT_RE =
  /^(\w+)\s*=\s*(person|softwareSystem|container|component)\s+"([^"]+)"(?:\s+"([^"]+)")?/;
const PROP_RE  = /^\s*"([^"]+)"\s+"([^"]+)"/;
const TAGS_RE  = /^\btags\b\s+((?:"[^"]+"\s*)+)/;
const TAG_START_RE = /\/\/\s*tag::([^[]+)\[\]/;
const TAG_END_RE   = /\/\/\s*end::[^[]+\[\]/;

export function parseProjectSlug(dsl: string): string {
  const m = dsl.match(/workspace\s+"([^"]+)"/);
  if (!m?.[1]) throw new Error('No workspace name found in DSL');
  return m[1].toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '');
}

export function parseElements(dsl: string): DslElement[] {
  const lines = dsl.split('\n');
  const results: DslElement[] = [];

  type ElementFrame = {
    kind: 'element';
    el: {
      id: string; name: string; type: C4ElementType;
      tagName: string | null; technology: string | undefined;
      tags: string[]; properties: Record<string, string>;
      slugChain: string[];
    };
  };
  type Frame =
    | { kind: 'workspace' | 'model' | 'other' }
    | ElementFrame
    | { kind: 'properties'; target: Record<string, string> };

  const stack: Frame[] = [];
  let pendingTagName: string | null = null;

  for (const rawLine of lines) {
    const line = rawLine.trim();

    const tagStart = line.match(TAG_START_RE);
    if (tagStart) { pendingTagName = tagStart[1] ?? null; continue; }
    if (TAG_END_RE.test(line)) { pendingTagName = null; continue; }

    const top = stack[stack.length - 1];

    if (top?.kind === 'properties') {
      const prop = line.match(PROP_RE);
      if (prop) top.target[prop[1]!] = prop[2]!;
      if (line === '}') stack.pop();
      continue;
    }

    const elMatch = line.match(ELEMENT_RE);
    if (elMatch) {
      const parentChain = (() => {
        for (let i = stack.length - 1; i >= 0; i--) {
          const f = stack[i];
          if (f?.kind === 'element') {
            const s = f.el.properties['slug'];
            return s ? [...f.el.slugChain, s] : f.el.slugChain;
          }
        }
        return [] as string[];
      })();

      const el: ElementFrame['el'] = {
        id: elMatch[1]!, name: elMatch[3]!, type: elMatch[2] as C4ElementType,
        technology: elMatch[4], tagName: pendingTagName,
        tags: [], properties: {}, slugChain: parentChain,
      };
      pendingTagName = null;

      if (line.endsWith('{')) stack.push({ kind: 'element', el });
      continue;
    }

    if (top?.kind === 'element') {
      const tagsMatch = line.match(TAGS_RE);
      if (tagsMatch) {
        top.el.tags = [...tagsMatch[1]!.matchAll(/"([^"]+)"/g)].map((m) => m[1]!);
      }
      if (line === 'properties {') {
        stack.push({ kind: 'properties', target: top.el.properties });
        continue;
      }
    }

    if (/^workspace\s/.test(line) && line.endsWith('{')) { stack.push({ kind: 'workspace' }); continue; }
    if (line === 'model {') { stack.push({ kind: 'model' }); continue; }
    if (line.endsWith('{') && !elMatch) { stack.push({ kind: 'other' }); continue; }

    if (line === '}') {
      const frame = stack.pop();
      if (frame?.kind === 'element') {
        const el = frame.el;
        const slug = el.properties['slug'];
        if (slug) {
          results.push({
            id: el.id, name: el.name, type: el.type,
            tagName: el.tagName, slug,
            elementPath: [...el.slugChain, slug].join('/'),
            technology: el.technology, tags: el.tags,
            properties: el.properties,
          });
        }
      }
    }
  }
  return results;
}

export function parseIndex(adocContent: string): ParsedIndex {
  const rawPhase = extractDocAttribute(adocContent, 'phase');
  if (rawPhase !== 'spec' && rawPhase !== 'impl') {
    throw new Error(`:phase: must be 'spec' or 'impl', got: ${rawPhase ?? '(missing)'}`);
  }
  const phase: Phase = rawPhase;
  const pairing = extractDocAttribute(adocContent, 'pairing');
  const dsl = extractDslBlock(adocContent);
  return { projectSlug: parseProjectSlug(dsl), phase, pairing, elements: parseElements(dsl) };
}
```

```typescript
// src/c4/validator.ts
import type { DslElement } from './types.js';

const STANDARD_ASSET_KEYS = new Set(['detail', 'bdd', 'tests', 'impl']);

export function validateSlugs(elements: DslElement[]): string[] {
  const byParent = new Map<string, DslElement[]>();
  for (const el of elements) {
    const slash = el.elementPath.lastIndexOf('/');
    const parent = slash === -1 ? '' : el.elementPath.slice(0, slash);
    const group = byParent.get(parent) ?? [];
    group.push(el);
    byParent.set(parent, group);
  }

  const errors: string[] = [];
  for (const [parent, siblings] of byParent) {
    const seen = new Set<string>();
    for (const el of siblings) {
      if (seen.has(el.slug)) {
        errors.push(
          `Duplicate slug "${el.slug}" under parent "${parent || '(root)'}"`
        );
      }
      seen.add(el.slug);
    }
  }
  return errors;
}

export function validateAssetKeys(elements: DslElement[]): string[] {
  const errors: string[] = [];
  for (const el of elements) {
    const standardKeys = Object.keys(el.properties).filter((k) =>
```

```typescript
// src/c4/lifecycle.ts
import { assetBranch } from './branch.js';
import type { PropertyLifecycleEvent } from './types.js';

function isAssetProp(key: string, value: string): boolean {
  return key !== 'slug' && value.endsWith('.adoc');
}

function stemOf(filename: string): string {
  return filename.replace(/\.adoc$/, '');
}

export function diffProperties(
  projectSlug: string,
  elementPath: string,
  oldProps: Record<string, string>,
  newProps: Record<string, string>,
  hasFinalCommit: (branch: string) => boolean
): PropertyLifecycleEvent[] {
  const events: PropertyLifecycleEvent[] = [];

  const oldAsset = new Map(
    Object.entries(oldProps).filter(([k, v]) => isAssetProp(k, v))
  );
  const newAsset = new Map(
    Object.entries(newProps).filter(([k, v]) => isAssetProp(k, v))
  );

  for (const [key, newFile] of newAsset) {
    const oldFile = oldAsset.get(key);
    if (!oldFile) {
      events.push({ type: 'spawn', elementSlug: elementPath, assetKey: key, assetFile: newFile });
    } else if (oldFile !== newFile) {
      const type = stemOf(oldFile) === stemOf(newFile)
        ? 'filename-change'
        : 'archive-old-spawn-new';
      events.push({ type, elementSlug: elementPath, assetKey: key, assetFile: newFile, oldAssetFile: oldFile });
    }
  }

  for (const [key, oldFile] of oldAsset) {
    if (!newAsset.has(key)) {
      const branch = assetBranch(projectSlug, elementPath, key);
      const type = hasFinalCommit(branch) ? 'archive-error' : 'defer';
      events.push({ type, elementSlug: elementPath, assetKey: key, assetFile: oldFile });
    }
  }

  return events;
}
```

```typescript
// src/c4/phase.ts
import { phaseTransitionTrailers } from './trailers.js';
import type { GitTrailers, Phase } from './types.js';

interface TransitionInput {
  currentPhase: Phase;
  requestedPhase: Phase;
  dslElement: string;
  dslTag: string;
  newPairing: string;
  knownPairings: string[];
}

type TransitionResult =
  | { ok: true; trailers: GitTrailers }
  | { ok: false; reason: string };

export function guardPhaseTransition(input: TransitionInput): TransitionResult {
  const { currentPhase, requestedPhase, dslElement, dslTag, newPairing, knownPairings } = input;

  if (currentPhase === requestedPhase) {
    return { ok: false, reason: `Phase is already "${currentPhase}"; no transition needed.` };
  }
  if (currentPhase === 'impl' && requestedPhase === 'spec') {
    return { ok: false, reason: `Cannot transition from impl back to spec.` };
  }
  if (!knownPairings.includes(newPairing)) {
    return { ok: false, reason: `Pairing "${newPairing}" is not in catalogue.` };
  }
  return { ok: true, trailers: phaseTransitionTrailers(dslElement, dslTag) };
}
```

```typescript
// src/c4/sandbox.ts
import type { TaskBlock } from './types.js';

export function parseTaskBlock(adocContent: string): TaskBlock {
  const m = adocContent.match(/\[task\]\n-{4,}\n([\s\S]*?)\n-{4,}/);
  if (!m?.[1]) {
    throw new Error(
      'Missing [task] block; writable_globs and readonly_globs are required for impl assets (R14.12)'
    );
  }
  const body = m[1];

  function extractList(section: string): string[] {
    const sm = body.match(new RegExp(`^${section}:\\s*\\n((?:\\s+-\\s+.+\\n?)*)`, 'm'));
    if (!sm?.[1]) return [];
    return sm[1]
      .split('\n')
      .map((l) => l.replace(/^\s*-\s*/, '').trim())
      .filter(Boolean);
  }

  return {
    writableGlobs: extractList('writable_globs'),
    readonlyGlobs: extractList('readonly_globs'),
  };
}
```

```typescript
// src/c4/extraction.ts

export function extractTaggedRegion(adocContent: string, tagName: string): string {
  const startMarker = `// tag::${tagName}[]`;
  const endMarker   = `// end::${tagName}[]`;

  const startIdx = adocContent.indexOf(startMarker);
  if (startIdx === -1) {
    throw new Error(`Tag "${tagName}" not found in index.adoc`);
  }
  const contentStart = startIdx + startMarker.length + 1;
  const endIdx = adocContent.indexOf(endMarker, contentStart);
  if (endIdx === -1) {
    throw new Error(`Tag "${tagName}" was opened but never closed in index.adoc`);
  }
  return adocContent.slice(contentStart, endIdx);
}
```

```typescript
// src/scaffold/worktree.ts
import { readFileSync } from 'node:fs';
import { join } from 'node:path';
import {
  parseIndex,
  validateSlugs,
  validateAssetKeys,
  projectIndexBranch,
  assetBranch,
  type DslElement,
  type ParsedIndex,
} from '../c4/index.js';

export interface C4WorktreeOptions {
  planDir: string;
}

export interface AssetBranchEntry {
  element: DslElement;
  assetKey: string;
  assetFile: string;
  branch: string;
}

export interface C4WorktreeResult {
  parsed: ParsedIndex;
  projectIndexBranch: string;
  assetBranches: AssetBranchEntry[];
}

export function loadC4Plan(options: C4WorktreeOptions): C4WorktreeResult {
  const indexPath = join(options.planDir, 'index.adoc');
  const adocContent = readFileSync(indexPath, 'utf-8');
  const parsed = parseIndex(adocContent);

  const errors = [...validateSlugs(parsed.elements), ...validateAssetKeys(parsed.elements)];
  if (errors.length > 0) {
    throw new Error(
      `C4 plan validation failed:\n${errors.map((e) => `  - ${e}`).join('\n')}`
    );
  }

  const assetBranches: AssetBranchEntry[] = parsed.elements.flatMap((element) =>
    Object.entries(element.properties)
      .filter(([key, value]) => key !== 'slug' && value.endsWith('.adoc'))
      .map(([key, file]) => ({
        element,
        assetKey: key,
        assetFile: file,
        branch: assetBranch(parsed.projectSlug, element.elementPath, key),
      }))
  );

  return {
    parsed,
    projectIndexBranch: projectIndexBranch(parsed.projectSlug),
    assetBranches,
  };
}
```
